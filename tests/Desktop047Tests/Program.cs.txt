void Check(bool ok, string message) { if (!ok) throw new Exception(message); }
string ChapterFor(Pack pack, string sectionId) => pack.Chapters.First(c => c.Sections.Any(s => s.Id == sectionId)).Id;

// 模拟桌面听书适配层（src/Listening047.cs）：待选支线 / 等待确认续接 / 死角都按 ResolveStop 倒计时结果处理（手动方式即“倒计时到点自动选择”）。
(List<string> Lines, int Stops, int Auto, List<string> Kinds) Listen(Pack pack, string sectionId, ListeningBranchPolicy policy, int number)
{
    var core = new CoreSession(pack, ChapterFor(pack, sectionId), policy);
    var lines = new List<string>(); var kinds = new List<string>(); int stops = 0, auto = 0, guard = 0;
    while (core.Current is { } item && guard++ < 100000)
    {
        if (item.Kind == ListeningItemKind.Choice || item.IsBlocking)
        {
            var stop = core.ResolveStop(number);
            if (stop == null) { if (item.SectionId == sectionId) stops++; break; }
            if (item.SectionId == sectionId) { auto++; kinds.Add(stop.Action.ToString()); if (stop.DeadEnd) stops++; }
            if (!core.ApplyStop(stop)) { stops++; break; }
            continue;
        }
        if (item.Kind == ListeningItemKind.Line && item.SectionId == sectionId) lines.Add(item.NodeId);
        if (!core.MoveNext()) { if (core.Current == item) break; }
    }
    return (lines, stops, auto, kinds);
}


// 模拟桌面游戏配音：玩家在游戏中点选与配音相同的项（单选项 / 探索菜单取 DesktopExploration 建议，普通菜单取默认第 n 项），统计路线句数、重复与无法继续处。
(int Lines, int Repeats, int Stops, int Menus, int Suggested, int Handovers, List<string> Read) Game(Pack pack, string sectionId, int number)
{
    // 未核出口（Gap）：与安卓游戏模式相同，自动播放交给玩家定位；模拟玩家按听书顺序定位到下一句（计为交手动，不计无法继续）。
    var order = Listen(pack, sectionId, number == 2 ? ListeningBranchPolicy.Second : ListeningBranchPolicy.First, number).Lines;
    int handovers = 0; string? lastLine = null;
    var e = new PlaybackEngine(pack); var start = pack.Chapters.SelectMany(c => c.Sections).First(s => s.Id == sectionId).StartId;
    if (!e.ConfirmGameLine(start) && !e.OpenGameMenu(start, sectionId)) throw new Exception("无法定位小节开头：" + e.NavigationError);
    var seen = new HashSet<string>(); var read = new List<string>(); int lines = 0, repeats = 0, stops = 0, menus = 0, suggested = 0;
    for (int guard = 0; guard < 20000; guard++)
    {
        var cur = e.Current; if (cur == null || cur.SectionId != sectionId) break;
        if (e.Mode == RunMode.Following && cur.Kind == "line") { lines++; read.Add(cur.Id); if (!seen.Add(cur.Id)) repeats++; lastLine = cur.Id; e.Next(); continue; }
        if (e.Mode == RunMode.Gap)
        {
            int at = lastLine == null ? -1 : order.IndexOf(lastLine);
            var target = order.Skip(at + 1).FirstOrDefault(id => !seen.Contains(id));
            if (target != null && e.ConfirmGameLine(target)) { handovers++; continue; }
        }
        if (e.Mode == RunMode.Choice)
        {
            menus++; var sug = DesktopExploration.Suggest(e, number);
            int index = sug?.Index ?? Math.Min(number - 1, e.AvailableOptions.Count - 1);
            if (sug != null) suggested++;
            DesktopExploration.Add(e, cur.Id, e.AvailableOptions[index].Id);
            string before = cur.Id; var mode = e.Mode; e.SelectBranch(index);
            if (e.CurrentId == before && e.Mode == mode) { Console.WriteLine($"   [menu stuck] {before} opt={e.AvailableOptions[index].Id} notice={e.Notice}"); stops++; break; }
            continue;
        }
        if (e.Mode == RunMode.Merge) { e.Next(); continue; }
        if (e.Mode == RunMode.End) break;
        Console.WriteLine($"   [game stop] mode={e.Mode} cur={cur.Id} kind={cur.Kind} next={cur.NextId} notice={e.Notice} err={e.NavigationError}");
        stops++; break;
    }
    return (lines, repeats, stops, menus, suggested, handovers, read);
}

// ---- 游戏配音终点规则（桌面文字跟随共用 GameEndWaitPolicy）：末句后不点击，提示终点文字等待；读到的文字在终点小节 + 下一小节里找，同文时只认下一小节首句 ----
PgrVoice.Following.GameEndWait EndWaitAt(PlaybackEngine e, string label)
{
    var decision = e.EvaluateCommonAutoPlayNext();
    Check(!(decision.Allowed && decision.Next != null), label + "：终点前竟允许自动点击 " + decision.Next?.Id);
    Check(!decision.Reason.Contains("共同剧情已结束"), label + "：仍是旧的结束提示 " + decision.Reason);
    Check(PgrVoice.Following.GameEndWaitPolicy.TryBegin(e, decision, out var wait) && wait != null, label + "：没有进入终点等待（" + decision.Code + "）");
    return wait!;
}
Node? TextLocate(Pack pack, PgrVoice.Following.GameEndWait wait, string text)
{
    // 与 MainWindow.MatchGameText 相同：终点小节 + 下一小节里按文字全等找台词，排除末句本身；多句同文只认下一小节首句。
    var exact = pack.Nodes.Where(n => n.Kind == "line" && n.Id != wait.SourceId && wait.SearchSections.Contains(n.SectionId) && n.Text == text).ToList();
    return PgrVoice.Following.GameEndWaitPolicy.Pick(pack, wait, exact);
}
void Resume(PlaybackEngine e, PgrVoice.Following.GameEndWait wait, Node next, string label)
{
    var source = e.Pack.ById[wait.SourceId];
    Check(TextLocate(e.Pack, wait, source.Text) == null || !PgrVoice.Following.GameEndWaitPolicy.CanResume(e, wait, source, out _, out _), label + "：画面仍是末句时结束了等待");
    var found = TextLocate(e.Pack, wait, next.Text);
    Check(found?.Id == next.Id, label + "：读到下一句却没有唯一定位到 " + next.Id + "（" + found?.Id + "）");
    Check(PgrVoice.Following.GameEndWaitPolicy.CanResume(e, wait, found!, out _, out var why), label + "：定位后仍不能续接：" + why);
    Check(e.ConfirmGameLine(found!.Id) && e.CurrentId == found.Id, label + "：导航器没有确认定位句");
}
