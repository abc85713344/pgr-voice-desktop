void Expect(bool condition, string why = "assertion") { if (!condition) throw new Exception(why); }
string Lines(ListeningSession s) => string.Join(",", s.Items.Where(i => i.Kind == ListeningItemKind.Line).Select(i => i.NodeId));
Node L(string id, string? next = null, string path = "", string section = "s") => new() { Id = id, Text = "台词" + id, SectionId = section, PathId = path, NextId = next };
ChoiceOption O(string id, string target, string? join = "m", string? path = null) => new() { Id = id, Label = "路线" + id, TargetId = target, PathId = path ?? id, MergeId = join, Verified = join != null };
void SkipNotices(ListeningSession s) { while(s.Current?.Kind==ListeningItemKind.Notice)s.MoveNext(); }
Node C(string id, params ChoiceOption[] options) => new() { Id = id, SectionId = "s", Kind = "choice", Options = options.ToList() };
Node M(string id, string? next = null) => new() { Id = id, SectionId = "s", Kind = "merge", NextId = next };
Node R(string id, string next, string path) => new() { Id = id, SectionId = "s", Kind = "return", PathId = path, NextId = next };
Pack P(params Node[] nodes)
{
    var p = new Pack { Id = "book", Root = Path.GetTempPath(), Nodes = nodes.ToList(),
        Chapters = new() { new() { Id = "ch", Title = "大章", Sections = new() { new() { Id = "s", Title = "第一小节", StartId = nodes[0].Id } } } } };
    p.Validate(); return p;
}

foreach(var policy in Enum.GetValues<ListeningBranchPolicy>())
{
    Test("互动遵循首项、全听或手动策略："+policy, () => {
        var p=Interaction();var game=new PlaybackEngine(p);var before=JsonSerializer.Serialize(game.ExportNavigation(),Json.Options);
        var s=new ListeningSession(p,"ch",policy);s.MoveNext();
        if(policy==ListeningBranchPolicy.Manual) {
            Expect(Lines(s)=="a");
            Expect(s.HasPendingChoice && s.Current!.NodeId=="c" && s.Current.Notice.StartsWith("等待互动："));
            Expect(!s.MoveNext() && !s.SeekNode("z"));Expect(!s.Choose("missing"));
        } else {
            Expect(!s.Items.Any(i=>i.Kind==ListeningItemKind.Choice));
            Expect(Lines(s)==(policy==ListeningBranchPolicy.All?"a,b,d,z":"a,b,z"));
            Expect(s.Current!.NodeId=="b");
        }
        var waiting=s.Capture();var reopened=new ListeningSession(p,"ch");Expect(reopened.TryRestore(waiting,out _));Expect(reopened.HasPendingChoice==(policy==ListeningBranchPolicy.Manual));
        if(policy==ListeningBranchPolicy.Manual) { Expect(reopened.Choose("two"));Expect(reopened.Current!.NodeId=="d");Expect(Lines(reopened)=="a,d,z"); }
        var chosen=reopened.Capture(321);var continued=new ListeningSession(p,"ch");
